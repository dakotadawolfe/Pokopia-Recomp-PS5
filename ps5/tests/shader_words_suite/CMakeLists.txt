cmake_minimum_required(VERSION 3.25)
project(pokopia_shader_words LANGUAGES CXX)
add_executable(ps5_shader_words_tests ../test_shader_words.cpp)
target_include_directories(ps5_shader_words_tests PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../runtime/include")
target_compile_features(ps5_shader_words_tests PRIVATE cxx_std_20)
if(NOT MSVC)
    target_compile_options(ps5_shader_words_tests PRIVATE -Wall -Wextra -Werror)
endif()
if(PS5)
    target_compile_definitions(ps5_shader_words_tests PRIVATE NXRT_TARGET_PS5=1)
    target_sources(ps5_shader_words_tests PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../app/thread_exit.cpp")
    target_link_options(ps5_shader_words_tests PRIVATE "LINKER:--wrap,pthread_create")
endif()
enable_testing()
add_test(NAME shader_words COMMAND ps5_shader_words_tests)
