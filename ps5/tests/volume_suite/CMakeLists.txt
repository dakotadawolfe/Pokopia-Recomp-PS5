cmake_minimum_required(VERSION 3.25)
project(pokopia_volume_tests LANGUAGES CXX)
set(root "${CMAKE_CURRENT_LIST_DIR}/../../..")
include("${root}/ps5/platform.cmake")
include("${root}/ps5/volume.cmake")
add_executable(ps5_volume_tests "${root}/ps5/tests/sync_main.cpp"
    "${root}/runtime/tests/test_svc_fs_volume.cpp")
target_include_directories(ps5_volume_tests PRIVATE "${root}/runtime/tests")
target_compile_options(ps5_volume_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(ps5_volume_tests PRIVATE nxrt_ps5_volume)
if(PS5)
    target_compile_definitions(ps5_volume_tests PRIVATE NXRT_TEST_PS5=1 NXRT_TEST_VOLUME=1)
    target_sources(ps5_volume_tests PRIVATE "${root}/ps5/app/thread_exit.cpp")
    target_link_options(ps5_volume_tests PRIVATE "LINKER:--wrap,pthread_create")
else()
    enable_testing()
    add_test(NAME save-volume COMMAND ps5_volume_tests)
    set_tests_properties(save-volume PROPERTIES TIMEOUT 120)
endif()
