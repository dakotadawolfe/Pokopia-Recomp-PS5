cmake_minimum_required(VERSION 3.25)
project(pokopia_time_tests LANGUAGES CXX)
set(root "${CMAKE_CURRENT_LIST_DIR}/../../..")
include("${root}/ps5/platform.cmake")
if(PS5)
    include("${root}/ps5/core.cmake")
    include("${root}/ps5/kernel.cmake")
    include("${root}/ps5/svc.cmake")
endif()
if(NOT TARGET nxrt_ps5_time_support)
    include("${root}/ps5/time.cmake")
endif()
add_executable(ps5_time_tests "${root}/ps5/tests/sync_main.cpp"
    "${root}/ps5/tests/test_time_clock.cpp" "${root}/ps5/tests/test_time_zone.cpp")
target_include_directories(ps5_time_tests PRIVATE "${root}/runtime/tests")
target_compile_options(ps5_time_tests PRIVATE -Wall -Wextra -Werror -mcx16)
target_link_libraries(ps5_time_tests PRIVATE nxrt_ps5_time_support nxrt_ps5_platform)
if(PS5)
    target_sources(ps5_time_tests PRIVATE "${root}/ps5/tests/test_time_ps5.cpp" "${root}/ps5/app/thread_exit.cpp")
    target_compile_definitions(ps5_time_tests PRIVATE NXRT_TEST_PS5=1 NXRT_TEST_TIME=1)
    target_link_libraries(ps5_time_tests PRIVATE nxrt_ps5_svc)
    target_link_options(ps5_time_tests PRIVATE "LINKER:--wrap,pthread_create")
else()
    enable_testing()
    add_test(NAME time COMMAND ps5_time_tests)
    set_tests_properties(time PROPERTIES TIMEOUT 60)
endif()
