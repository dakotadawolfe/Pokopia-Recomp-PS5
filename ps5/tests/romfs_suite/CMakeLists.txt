cmake_minimum_required(VERSION 3.25)
project(pokopia_romfs_tests LANGUAGES CXX)
set(root "${CMAKE_CURRENT_LIST_DIR}/../../..")
include("${root}/ps5/platform.cmake")
include("${root}/ps5/romfs.cmake")
add_executable(ps5_romfs_tests "${root}/ps5/tests/sync_main.cpp"
    "${root}/runtime/tests/test_svc_fs_romfs.cpp")
target_include_directories(ps5_romfs_tests PRIVATE "${root}/runtime/tests")
target_compile_options(ps5_romfs_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(ps5_romfs_tests PRIVATE nxrt_ps5_romfs)
if(PS5)
    target_compile_definitions(ps5_romfs_tests PRIVATE NXRT_TEST_PS5=1 NXRT_TEST_ROMFS=1)
    target_sources(ps5_romfs_tests PRIVATE "${root}/ps5/app/thread_exit.cpp")
    target_link_options(ps5_romfs_tests PRIVATE "LINKER:--wrap,pthread_create")
else()
    target_compile_definitions(ps5_romfs_tests PRIVATE NXRT_ROMFS_NO_LSTAT=1)
    target_link_options(ps5_romfs_tests PRIVATE "LINKER:--wrap,lstat")
    enable_testing()
    add_test(NAME romfs COMMAND ps5_romfs_tests)
    set_tests_properties(romfs PROPERTIES TIMEOUT 120)
endif()
