cmake_minimum_required(VERSION 3.25)
project(pokopia_picker_tests LANGUAGES CXX)
set(root "${CMAKE_CURRENT_LIST_DIR}/../../..")
include("${root}/ps5/platform.cmake")
include("${root}/ps5/picker.cmake")
add_executable(ps5_picker_tests "${root}/ps5/tests/sync_main.cpp" "${root}/ps5/tests/test_folder_picker.cpp")
target_include_directories(ps5_picker_tests PRIVATE "${root}/runtime/tests")
target_compile_options(ps5_picker_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(ps5_picker_tests PRIVATE nxrt_ps5_picker nxrt_ps5_platform)
if(PS5)
    target_compile_definitions(ps5_picker_tests PRIVATE NXRT_TEST_PS5=1)
else()
    target_compile_definitions(ps5_picker_tests PRIVATE NXRT_PICKER_NO_REALPATH=1)
    target_link_options(ps5_picker_tests PRIVATE "LINKER:--wrap,realpath" "LINKER:--wrap,lstat")
    enable_testing()
    add_test(NAME folder_picker COMMAND ps5_picker_tests)
endif()
